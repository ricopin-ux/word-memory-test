<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>7-Second Word Display</title>
    <!-- Tailwind CSSを読み込みます -->
    <script src="https://cdn.tailwindcss.com"></script>
    <style>
        /* Interフォントを使用します */
        body {
            font-family: "Inter", sans-serif;
        }
    </style>
</head>
<body class="bg-gray-100 flex items-center justify-center min-h-screen">

    <div class="bg-white p-12 rounded-2xl shadow-xl text-center max-w-lg w-full">
        <h1 class="text-2xl font-semibold text-gray-700 mb-6">Word of the Moment</h1>
        
        <!-- このdivに単語が表示されます -->
        <div id="word-display" class="text-6xl font-bold text-blue-600 h-24 flex items-center justify-center transition-opacity duration-500">
            <!-- 最初の単語はJSによって読み込まれます -->
        </div>
    </div>

    <script>
        // 10個の英単語のリスト
        const words = [
            "Serendipity",
            "Ephemeral",
            "Luminous",
            "Tranquility",
            "Mellifluous",
            "Sonder",
            "Petrichor",
            "Resilience",
            "Cacophony",
            "Solitude"
        ];

        let currentIndex = 0;
        const wordDisplayElement = document.getElementById('word-display');

        function showNextWord() {
            if (currentIndex < words.length) {
                // フェードアウトエフェクトを適用します
                wordDisplayElement.classList.add('opacity-0');

                // フェードアウトが完了するのを待ってから単語を変更します
                setTimeout(() => {
                    // 単語を更新します
                    wordDisplayElement.textContent = words[currentIndex];
                    // フェードインエフェクトを適用します
                    wordDisplayElement.classList.remove('opacity-0');
                    
                    // 次の単語のインデックスに移動します
                    currentIndex++;
                }, 500); // フェードアウトのための500ms

            } else {
                // すべての単語が表示されました
                wordDisplayElement.textContent = "Done!";
                wordDisplayElement.classList.remove('opacity-0');
                
                // インターバルを停止します
                if (wordInterval) {
                    clearInterval(wordInterval);
                }
            }
        }

        // 読み込み時に最初の単語をすぐに表示します
        showNextWord();

        // 7秒ごと（7000ミリ秒）に次の単語を表示するインターバルを設定します
        const wordInterval = setInterval(showNextWord, 7000);

    </script>

</body>
</html>
